// ---------- 5. MAIN CALCULATOR LOOP ----------
window.runMasterplanCalculator = function(){
    var codeKey = $('calc-code').value || 'rak_sub';
    var P = PROFILES[codeKey] || PROFILES.rak_sub;
    var u = $('calc-unit').value;
    var mode = $('calc-mode').value;
    var plot = Math.max(50, parseFloat($('calc-plot-size').value) || 900);
    var hh = Math.max(1, parseFloat($('calc-hh-size').value) || P.hh);
    var far = Math.max(0.1, parseFloat($('calc-far').value) || 1.2);
    var fl = Math.max(0.5, parseFloat($('calc-floors').value) || 2);
    var gm = $('calc-gfa-mode').value;
    var ratio = Math.min(100, Math.max(1, parseFloat($('calc-cover').value) || 65)) / 100;
    var V = function(id, d){ var e = $(id); return e ? (parseFloat(e.value) || 0) : d; };

    var Pct = {
        res: +$('calc-pct-res').value || 0,
        roads: +$('calc-pct-roads').value || 0,
        open: +$('calc-pct-open').value || 0,
        fac: +$('calc-pct-fac').value || 0,
        com: +$('calc-pct-com').value || 0
    };
    ['res','roads','open','fac','com'].forEach(function(k){ $('lbl-pct-'+k).textContent = Pct[k] + '%'; });
    var sum = Pct.res + Pct.roads + Pct.open + Pct.fac + Pct.com;
    var b = $('calc-total-pct-badge');
    if(sum === 100){
        b.className = 'text-xs font-mono px-2.5 py-0.5 rounded bg-emerald-500/20 text-emerald-300 border border-emerald-500/30';
        b.textContent = 'المجموع: 100% (متوازن ✓)';
    } else {
        b.className = 'text-xs font-mono px-2.5 py-0.5 rounded bg-rose-500/20 text-rose-300 border border-rose-500/40';
        b.textContent = 'المجموع: ' + sum + '% (' + (sum > 100 ? 'زيادة ' + (sum - 100) : 'نقص ' + (100 - sum)) + '%)';
    }

    var m2, plots, pop, popEm, popMix, totalUnits, aptUnits = 0, derived = false;
    var mixOn = P.isRefMan && $('ca-mix-on').checked;

    if(mode === 'pop' && !mixOn){
        var pin = Math.max(0, parseFloat($('calc-pop-input').value) || 0);
        plots = Math.ceil(pin / hh - 1e-9);
        pop = Math.round(pin);
        m2 = Math.max(100, plots * plot / (Math.max(Pct.res, 1) / 100));
        derived = true;
    } else {
        var v = parseFloat($('calc-area-input').value) || 0;
        m2 = Math.max(100, u === 'ha' ? v * 10000 : v);
    }
    var ha = m2 / 10000;
    $('calc-area-ha').value = ha;

    var M = {};
    ['res','roads','open','fac','com'].forEach(function(k){ M[k] = m2 * Pct[k] / 100; });

    if(mixOn){
        var vInv = Math.max(0, V('ca-v-inv', 0));
        aptUnits = Math.max(0, V('ca-apts', 0));
        if(codeKey === 'rak_urb'){

            var vPriv = u1 + u2 + u3 + u4;
            popEm = Math.round(u1*2.5 + u2*3.5 + u3*5.1 + u4*6.57);
            popMix = Math.round(vInv*4.39 + aptUnits*2.9);
            plots = vPriv + vInv;
            totalUnits = vPriv + vInv + aptUnits;
        } else {
            var vPriv2 = Math.max(0, V('ca-v-priv', 0));
            var useUpd = (codeKey === 'rak_sub' && $('ca-sub-demo').value === 'upd');

            popMix = Math.round(vInv * invF + aptUnits * aptF);

            totalUnits = vPriv2 + vInv + aptUnits;
        }
        pop = popEm + popMix;
    } else {
        if(!derived){
            plots = Math.floor(M.res / plot + 1e-9);
            pop = Math.round(plots * hh);
        }
        totalUnits = plots;
        var pType = P.isRak ? $('rak-parcel-type').value : 'custom';
        if(pType === 'rak_inv' || pType === 'rak_apt'){
            popEm = 0; popMix = pop;
            if(pType === 'rak_apt') aptUnits = plots;
        } else {
            popEm = pop; popMix = 0;
        }
    }

    $('calc-area-m2-label').textContent = (derived ? 'مساحة مشتقة: ' : '') + N(m2) + ' م² = ' + ha.toLocaleString('en', {maximumFractionDigits: 2}) + ' هكتار';

    // Setbacks & Envelope
    var w = Math.max(1, V('calc-plot-w', 30)), f = V('cs-f', 2), sb = V('cs-s', 1.5), rr = V('cs-r', 1.5);
    var dd = plot / w, fw = w - 2*sb, fd = dd - f - rr;
    var envA = (fw > 0 && fd > 0) ? fw * fd : 0;
    var envF = Math.min(1, envA / plot);
    var cov = Math.min(envF, ratio);

    var resPark = Math.round(totalUnits * (parseFloat($('calc-park').value) || 0));

    var musallaWP = Math.round(comPark * 0.35);

    $('out-green-per-capita').textContent = pop > 0 ? (M.open / pop).toFixed(1) : '0';
    $('out-foot').textContent = A(foot).replace(/ (م²|هكتار)$/, '');
    $('out-foot-s').textContent = (u === 'ha' ? 'هكتار' : 'م²') + ' = ' + (cov*100).toFixed(0) + '% من القسيمة';
    $('out-floors').textContent = floors.toFixed(1);
    $('out-floors-s').textContent = gm === 'floors' ? 'الأدوار المدخلة' : 'مكافئ من FAR';
    $('out-env').textContent = (envF * 100).toFixed(0) + '%';
    $('out-env-s').textContent = 'حدود بناء ' + (fw>0?fw.toFixed(1):0) + '×' + (fd>0?fd.toFixed(1):0) + ' م';

        ['الإسكان الصافي (Net Residential)','res','48% – 55% (50% صافي قسائم)','text-amber-300'],

    var kvaCap = Math.max(100, g('kva') || 1000);
    var trf = pop > 0 ? Math.max(1, Math.ceil(kva / kvaCap)) : 0;
    var waste = pop * g('waste') / 1000;
    var wD = pop * (wL + (wH - wL)*lv) / 1000;

    $('ut-w').textContent = N(pop*wL/1000) + '–' + N(pop*wH/1000) + ' م³/يوم';
    $('ut-w-s').textContent = 'المعتمد ('+lvLbl+') ' + N(wD) + ' · ذروة ×1.5 ≈ ' + N(wD*1.5);
    $('ut-s').textContent = N(pop*wL/1000*sewF) + '–' + N(pop*wH/1000*sewF) + ' م³/يوم';
    $('ut-s-s').textContent = 'المعتمد ' + N(wD*sewF) + ' م³/يوم';
    $('ut-p').textContent = (kDesign/1000).toFixed(2) + ' م.و (' + trf + ' محول)';
    $('ut-p-s').textContent = lvLbl + ' × تباين ' + divF.toFixed(2) + ' ≈ ' + N(kva) + ' ك.ف.أ (نطاق ' + (kLo/1000).toFixed(1) + '–' + (kHi/1000).toFixed(1) + ' م.و)';
    $('ut-g').textContent = waste.toFixed(1) + ' طن/يوم';
    $('ut-g-s').textContent = 'للتجميع والنقل اليومي';
    if($('ut-summary-chip')) $('ut-summary-chip').textContent = 'المياه: ' + N(wD) + ' م³/يوم · الكهرباء: ' + (kDesign/1000).toFixed(2) + ' م.و (' + trf + ' محول)';

    // ---------- 6. DETECT PLANNING SCALE & UPDATE HIERARCHY BAR ----------
    var det = detectPlanningScale(codeKey, ha, pop, derived);
    var activeScaleNum = HIER_STATE.scaleMode === 'auto' ? det.tier : (HIER_STATE.scaleMode === 'all' ? 4 : parseInt(HIER_STATE.scaleMode, 10));
    var showAll = (HIER_STATE.scaleMode === 'all');

    // Sync checkboxes & buttons
    $('hchk-noncenter').checked = HIER_STATE.includeNonCenter;
    $('hchk-hidezero').checked = HIER_STATE.hideZero;
    ['auto','1','2','3','4','all'].forEach(function(k){
        var el = $('hbtn-'+k); if(!el) return;
        var active = (HIER_STATE.scaleMode === k);
        el.className = 'h-scale-btn px-2.5 py-1.5 rounded-lg text-xs font-bold border transition ' +
            (active ? 'bg-teal-600 text-white border-teal-400 shadow' : 'bg-slate-800 text-slate-300 border-slate-700 hover:bg-slate-700');
    });
    ['cumulative','exact'].forEach(function(k){
        var el = $('hscope-'+k); if(!el) return;
        var active = (HIER_STATE.scopeMode === k);
        el.className = 'px-2.5 py-1 rounded-lg font-bold border transition ' +
            (active ? 'bg-amber-500/25 text-amber-300 border-amber-500/50' : 'bg-slate-800 text-slate-400 border-slate-700 hover:bg-slate-700');
    });

    // Detected Banner
    var detMeta = TIER_META[det.tier];
    var estClusters = Math.max(1, Math.round(ha / 38.5));
    var estNH = Math.max(1, Math.round(pop / 5000));
    $('hier-detected-banner').innerHTML =
        '<div class="space-y-0.5">' +
            '<div class="flex flex-wrap items-center gap-2">' +
                '<span class="text-xs font-bold text-slate-300">التصنيف التلقائي لموقعك:</span>' +
                '<span class="px-2.5 py-0.5 rounded-full text-xs font-extrabold border '+detMeta.badge+'"><i class="fa-solid '+detMeta.icon+' ml-1"></i>'+detMeta.ar+' ('+detMeta.en+')</span>' +
                (HIER_STATE.scaleMode !== 'auto' ? '<span class="px-2 py-0.5 rounded bg-amber-500/20 text-amber-300 text-[10px] font-bold">عرض يدوي: '+(showAll?'الكل':TIER_META[activeScaleNum].ar)+'</span>' : '') +
            '</div>' +
            '<div class="text-[11px] text-slate-400">'+det.desc+'</div>' +
            (det.mismatch ? '<div class="text-[11px] text-amber-300 pt-0.5"><i class="fa-solid fa-circle-info ml-1"></i>'+det.mismatch+'</div>' : '') +
        '</div>' +
        '<div class="text-[11px] font-mono text-teal-300 bg-slate-950/70 px-3 py-1.5 rounded-lg border border-slate-800 shrink-0">' +
            'تقدير البنية: ≈ ' + estClusters + ' مجموعة سكنية · ' + (det.tier >= 2 ? (estNH + ' مجاورة') : 'أقل من مجاورة') +
        '</div>';

    // 4-Step Visual Stepper
    $('hier-stepper').innerHTML = [1, 2, 3, 4].map(function(t){
        var tm = TIER_META[t];
        var isDetected = (t === det.tier);
        var isIncluded = showAll ? true : (HIER_STATE.scopeMode === 'exact' ? (t === activeScaleNum) : (t <= activeScaleNum));
        var cls = isIncluded
            ? ('cursor-pointer p-2 rounded-lg border font-bold transition ' + tm.hdr + (isDetected ? ' ring-1 ring-amber-400' : ''))
            : 'cursor-pointer p-2 rounded-lg border border-slate-800 bg-slate-950/40 text-slate-500 opacity-55 hover:opacity-90 transition';
        return '<div onclick="setHierScale(\''+t+'\')" class="'+cls+'">' +
            '<div><i class="fa-solid '+tm.icon+' ml-1"></i>'+tm.ar+'</div>' +
            '<div class="text-[10px] opacity-80 font-mono mt-0.5">'+(isDetected ? '★ مستوى موقعك' : (isIncluded ? 'مشمول بالعرض' : 'خارج نطاق الموقع'))+'</div>' +
        '</div>';
    }).join('');

    // Helper: does tier pass filter?
    function tierVisible(t){
        if(t === 0) return true; // Utilities always shown in its own clean group
        if(showAll) return true;
        return HIER_STATE.scopeMode === 'exact' ? (t === activeScaleNum) : (t <= activeScaleNum);
    }

    // ---------- 7. RENDER GROUPED HIERARCHICAL SERVICES ----------
    var allRows = [];
    var isEditableLegacy = !P.isRefMan;
    var infillBox = $('ca-infill-box');
    var pillsEl = $('ca-summary-pills');
    infillBox.classList.add('hidden');
    pillsEl.innerHTML = '';

    if(P.isRefMan){
        $('sv-section-title').textContent = 'رابعاً: خدمات ومرافق «' + (showAll ? 'جميع المستويات' : TIER_META[activeScaleNum].ar) + '» — ' + P.n;
        $('sv-section-sub').textContent = (HIER_STATE.scopeMode === 'cumulative' && !showAll)
            ? 'وضع تراكمي: يعرض خدمات المستوى المحدد مضافاً إليها خدمات المستويات الأدنى المندمجة داخله.'
            : (showAll ? 'يعرض كافة المستويات التخطيطية من المجموعة السكنية حتى القطاع الشامل.' : 'وضع معزول: يعرض الخدمات المضافة على هذا المستوى فقط دون المستويات الأدنى.');

        var cfRes = computeRefFacilities(codeKey, ha, popEm, popMix, totalUnits, aptUnits, trf, V('ca-agri-m2', 0), $('ca-cd-risk') ? $('ca-cd-risk').value : 'C');
        allRows = cfRes.rows;

        pillsEl.innerHTML =
            '<span class="px-2 py-1 rounded bg-slate-900 border border-slate-700 text-teal-300">المصلون (35%): '+N(cfRes.worshipers)+'</span>' +
            '<span class="px-2 py-1 rounded bg-slate-900 border border-slate-700 text-amber-300">الطلاب: '+N(cfRes.students)+'</span>' +
            '<span class="px-2 py-1 rounded bg-slate-900 border border-slate-700 text-emerald-300">الحدائق: '+(cfRes.openSpaceM2/m2*100).toFixed(1)+'%</span>' +
            '<span class="px-2 py-1 rounded bg-slate-900 border border-slate-700 text-sky-300">مصليات التجاري: '+musallaWP+' م²</span>';

        if(pop > 0 && pop < 3000){
            infillBox.classList.remove('hidden');
            infillBox.innerHTML =
                '<b><i class="fa-solid fa-circle-info ml-1"></i> تنبيه الإحلال والتجمعات الصغيرة (&lt; 3,000 نسمة — نهج نصيب الفرد):</b> ' +
                'ينطبق نهج <b>نصيب الفرد (Per Capita GFA)</b>: عيادات = <b>'+(pop*0.50).toFixed(0)+' م²</b> (أدنى 3,000 م²) · تجاري = <b>'+(pop*0.80).toFixed(0)+' م²</b> · مركز مجتمعي = <b>'+(pop*0.13).toFixed(0)+' م²</b> · حضانة = <b>'+(pop*0.12).toFixed(0)+' م²</b>.';
        }
    } else {
        $('sv-section-title').textContent = 'رابعاً: خدمات ومرافق «' + (showAll ? 'جميع المستويات' : TIER_META[activeScaleNum].ar) + '» — ' + P.n;
        $('sv-section-sub').textContent = 'الخدمات مقسّمة هرمياً حسب مستوى الموقع، وقابلة للتعديل المباشر في خانات (نسمة/وحدة) و(مساحة الوحدة).';
        var baseList = EG_SV;
        var edits = getLegacyEdits(codeKey);
        allRows = baseList.map(function(r, idx){
            var ed = edits[idx] || {};
            var per = r[4] ? (ed.per || r[4]) : 0;
            var ar  = ed.area || r[5];
            var fr  = per ? pop / per : 1;
            var cnt = pop <= 0 ? 0 : (per ? (fr >= 0.5 ? Math.ceil(fr) : 0) : trf);
            return { tier:r[0], cat:r[1], tag:r[2], name:r[3], basis:r[6]||'', per:per, landU:ar, gfaU:0, cnt:cnt, origIdx:idx };
        });
    }

    // Group rows by tier order [1, 2, 3, 4, 0]
    var order = [1, 2, 3, 4, 0];
    var groupsHtml = '';
    var grandLand = 0, grandGfa = 0, grandCount = 0;
    var hiddenTiers = [];

    order.forEach(function(t){
        var tierRowsAll = allRows.filter(function(r){ return r.tier === t; });
        if(tierRowsAll.length === 0) return;

        if(!tierVisible(t)){
            if(t > 0) hiddenTiers.push(TIER_META[t].ar);
            return;
        }

        var filtered = tierRowsAll.filter(function(r){
            if(!HIER_STATE.includeNonCenter && r.cat === 'noncenter') return false;
            if(HIER_STATE.hideZero && r.cnt <= 0) return false;
            return true;
        });

        var tm = TIER_META[t];
        var gLand = 0, gGfa = 0, gCnt = 0;
        filtered.forEach(function(r){ gLand += r.cnt * r.landU; gGfa += r.cnt * (r.gfaU || 0); gCnt += r.cnt; });

        
        grandLand += gLand; grandGfa += gGfa; grandCount += gCnt;

        var tblHead = isEditableLegacy
            ? '<tr><th class="p-2">التصنيف</th><th class="p-2">الخدمة / المرفق</th><th class="p-2">نسمة/وحدة</th><th class="p-2">مساحة الوحدة (م²)</th><th class="p-2 font-mono">العدد</th><th class="p-2 font-mono">إجمالي الأرض</th></tr>'
            : '<tr><th class="p-2">التصنيف</th><th class="p-2">الخدمة / المرفق</th><th class="p-2">المعيار / النطاق</th><th class="p-2 font-mono">العدد</th><th class="p-2 font-mono">أرض/وحدة</th><th class="p-2 font-mono">إجمالي الأرض</th><th class="p-2 font-mono">إجمالي GFA</th></tr>';

        var rowsHtml = filtered.length === 0
            ? '<tr><td colspan="7" class="p-3 text-center text-slate-400 text-xs">لا توجد خدمات مطلوبة (&gt; 0) ضمن هذا المستوى حالياً — ألغِ «إظهار المطلوب فقط» لرؤية كافة المعايير.</td></tr>'
            : filtered.map(function(r){
                var tLand = r.cnt * r.landU;
                var tGfa  = r.cnt * (r.gfaU || 0);
                var rowCls = r.cnt > 0 ? '' : 'opacity-45';
                var typeChip = r.cat === 'noncenter'
                    ? '<span class="px-1.5 py-0.5 rounded bg-amber-500/15 text-amber-300 border border-amber-500/30 text-[10px]">'+r.tag+'</span>'
                    : '<span class="px-1.5 py-0.5 rounded bg-slate-800 text-slate-300 border border-slate-700 text-[10px]">'+r.tag+'</span>';
                if(isEditableLegacy){
                    return '<tr class="'+rowCls+'">' +
                        '<td class="p-2">'+typeChip+'</td>' +
                        '<td class="p-2 font-bold text-white">'+r.name+(r.basis?'<div class="text-[10px] font-normal text-slate-400">'+r.basis+'</div>':'')+'</td>' +
                        '<td class="p-2">'+(r.per ? '<input type="number" min="1" value="'+r.per+'" onchange="updateLegacyCell(\''+codeKey+'\','+r.origIdx+',\'per\',this.value)" class="w-20 rounded bg-slate-900 border border-slate-700 px-2 py-1 font-mono text-xs text-white">' : '<span class="text-slate-400 text-[11px]">حسب الحمل</span>')+'</td>' +
                        '<td class="p-2"><input type="number" min="1" value="'+r.landU+'" onchange="updateLegacyCell(\''+codeKey+'\','+r.origIdx+',\'area\',this.value)" class="w-20 rounded bg-slate-900 border border-slate-700 px-2 py-1 font-mono text-xs text-white"></td>' +
                        '<td class="p-2 font-mono font-bold '+(r.cnt>0?'text-amber-300':'text-slate-500')+'">'+r.cnt+'</td>' +
                        '<td class="p-2 font-mono text-teal-300">'+(tLand>0?A(tLand):'—')+'</td>' +
                    '</tr>';
                }
                return '<tr class="'+rowCls+'">' +
                    '<td class="p-2">'+typeChip+'</td>' +
                    '<td class="p-2 font-bold text-white">'+r.name+'</td>' +
                    '<td class="p-2 text-[11px] text-slate-400">'+r.basis+'</td>' +
                    '<td class="p-2 font-mono font-bold '+(r.cnt>0?'text-amber-300':'text-slate-500')+'">'+r.cnt+'</td>' +
                    '<td class="p-2 font-mono">'+(r.landU>0?N(r.landU)+' م²':'مدمج')+'</td>' +
                    '<td class="p-2 font-mono text-teal-300">'+(tLand>0?A(tLand):'—')+'</td>' +
                    '<td class="p-2 font-mono text-sky-300">'+(tGfa>0?N(tGfa)+' م²':'—')+'</td>' +
                '</tr>';
            }).join('');

        groupsHtml +=
            '<details open class="um-pin-det rounded-xl border border-slate-700 overflow-hidden bg-slate-900/70">' +
                '<summary class="cursor-pointer px-3.5 py-2.5 border-b flex flex-wrap items-center justify-between gap-2 text-xs sm:text-sm font-extrabold list-none '+tm.hdr+'">' +
                    '<span><i class="fa-solid '+tm.icon+' ml-1.5"></i>'+tm.ar+' <span class="font-mono text-[11px] opacity-80">('+tm.en+')</span></span>' +
                    '<span class="flex flex-wrap items-center gap-2 font-mono text-[11px]">' +
                        '<span class="px-2 py-0.5 rounded bg-slate-950/60 border border-slate-700">العدد: '+gCnt+'</span>' +
                        '<span class="px-2 py-0.5 rounded bg-slate-950/60 border border-slate-700">الأرض: '+A(gLand)+' ('+(gLand/m2*100).toFixed(1)+'%)</span>' +
                        (gGfa > 0 ? '<span class="px-2 py-0.5 rounded bg-slate-950/60 border border-slate-700">GFA: '+N(gGfa)+' م²</span>' : '') +
                    '</span>' +
                '</summary>' +
                '<div class="overflow-x-auto"><table class="w-full text-right text-xs"><thead class="bg-slate-800/90 text-slate-300">'+tblHead+'</thead><tbody class="divide-y divide-slate-800/80">'+rowsHtml+'</tbody></table></div>' +
            '</details>';
    });


    // Grand Total Summary Bar
    groupsHtml +=
        '<div class="p-3.5 rounded-xl bg-gradient-to-l from-slate-900 to-teal-950/50 border border-teal-500/40 flex flex-wrap items-center justify-between gap-3 text-xs sm:text-sm font-extrabold">' +
            '<span class="text-amber-300"><i class="fa-solid fa-sigma ml-1"></i> إجمالي الخدمات والمرافق المعروضة ('+(grandLand/m2*100).toFixed(1)+'% من مساحة الموقع)</span>' +
            '<div class="flex flex-wrap items-center gap-3 font-mono">' +
                '<span class="text-white">الوحدات: '+grandCount+'</span>' +
                '<span class="text-teal-300">الأرض: '+A(grandLand)+'</span>' +
                (grandGfa > 0 ? '<span class="text-sky-300">GFA: '+N(grandGfa)+' م²</span>' : '') +
            '</div>' +
        '</div>';

    $('sv-groups-container').innerHTML = groupsHtml;

    // Hidden Higher Tiers Note
    var hiddenEl = $('sv-hidden-note');
    if(hiddenTiers.length > 0){
        hiddenEl.classList.remove('hidden');
        hiddenEl.innerHTML =
            '<div class="flex flex-wrap items-center justify-between gap-2">' +
                '<span><i class="fa-solid fa-eye-slash text-amber-400 ml-1.5"></i> مستويات غير معروضة حالياً (أعلى من أو خارج نطاق التصفية): <b class="text-amber-300">'+hiddenTiers.join(' · ')+'</b> — تُخدم عادةً على مستوى تخطيطي أعلى خارج حدود موقعك.</span>' +
                '<button onclick="setHierScale(\'all\')" class="px-2.5 py-1 rounded-lg bg-slate-800 hover:bg-slate-700 text-teal-300 border border-slate-700 text-[11px] font-bold shrink-0">عرض جميع المستويات</button>' +
            '</div>';
    } else {
        hiddenEl.classList.add('hidden');
    }

    // Budget vs Required Balance
    var budgetCompare = P.isRefMan ? (M.fac + M.open) : M.fac;
    var diff = budgetCompare - grandLand;
    $('sv-bal').innerHTML = '<div class="p-3 rounded-xl border text-xs '+(diff>=0?'bg-teal-950/50 border-teal-500/40 text-teal-200':'bg-rose-950/50 border-rose-500/40 text-rose-200')+'">' +
        'المخصص في ميزانيتك ('+(P.isRefMan ? ('خدمات '+A(M.fac)+' + حدائق '+A(M.open)+' = <b>'+A(budgetCompare)+'</b>') : ('خدمات <b>'+A(budgetCompare)+'</b>'))+') مقابل المطلوب ضمن المستوى المعروض (<b>'+A(grandLand)+'</b>): ' +
        (diff>=0 ? '<b>فائض '+A(diff)+'</b> يغطي احتياج المستوى بالكامل.' : '<b>عجز '+A(-diff)+'</b> — ارفع نسب الخدمات/الحدائق أو راجع المرافق القائمة بالمحيط.') +
    '</div>';

    // Code D Utility Plot Reference Footer
    $('util-note').innerHTML = P.isRak
        ? '<b class="text-amber-300"><i class="fa-solid fa-bolt ml-1"></i> مرجع أبعاد قطع المرافق (قيم مرجعية عامة):</b> محطة توزيع DSS = <b>5×9م (45 م²)</b> · رئيسية PSS = <b>100×80م</b> · شبكة Grid = <b>300×300م</b> · رفع صرف صحي = <b>50×50م</b> · تصريف أمطار = <b>50×50م</b> · غاز/إطفاء = <b>30×30م</b> · خزان مياه = <b>100×100م</b>.'
        : '<b class="text-sky-300">مرجع المرافق:</b> تُنسق مواقع المحولات والخزانات ومحطات الرفع مع شركات المرافق المختصة بالمحافظة.';

    // Diagnostics Alerts
    var al = [];
    if(sum !== 100) al.push(['rose','fa-triangle-exclamation','مجموع نسب الاستعمالات '+sum+'% وليس 100%. اضغط زر «موازنة 100%».']);
    if(Pct.roads > 28) al.push(['amber','fa-road-circle-exclamation','نسبة الطرق ('+Pct.roads+'%) مرتفعة (> 28%) وترفع تكلفة البنية التحتية والجزر الحرارية.']);
    else if(Pct.roads < 15) al.push(['amber','fa-road','نسبة الطرق ('+Pct.roads+'%) منخفضة وقد لا تكفي لحرم الشوارع والمواقف والمرافق.']);
    if(Pct.open < 8) al.push(['rose','fa-tree','نسبة المناطق المفتوحة ('+Pct.open+'%) أقل من الموصى به (10–12%).']);
    if(envF <= 0) al.push(['rose','fa-ruler-combined','الارتدادات المدخلة تلتهم القسيمة بالكامل!']);
    if(P.isRak){
        al.push(['teal','fa-mosque','مواقف التجاري ('+N(comPark)+' موقف بمعدل 1/70م² للمحال والمكاتب؛ وللمطاعم 1/50م²) ← تستوجب مصليات  بإجمالي <b>'+musallaWP+' م²</b> ('+Math.ceil(musallaWP/2)+'م² رجال + '+Math.ceil(musallaWP/2)+'م² نساء).']);
    }
    al.push(['teal','fa-bolt','ملخص الموقع ('+N(m2)+' م² = '+ha.toFixed(2)+' هكتار · مستوى '+detMeta.ar+'): '+N(pop)+' نسمة · '+N(totalUnits)+' وحدة · عدد المحولات ('+lvLbl+' × تباين '+divF+'): <b>'+trf+' محول</b> بمساحة DSS = '+N(trf*P.dssArea)+' م².']);

    var sm = {rose:'bg-rose-950/50 border-rose-500/40 text-rose-200', amber:'bg-amber-950/50 border-amber-500/40 text-amber-200', teal:'bg-teal-950/50 border-teal-500/40 text-teal-200'};
    $('calc-diagnostics').innerHTML = al.map(function(a){
        return '<div class="p-3 rounded-xl border '+sm[a[0]]+' text-xs flex items-start gap-2.5"><i class="fa-solid '+a[1]+' mt-0.5 text-sm"></i><span>'+a[2]+'</span></div>';
    }).join('');
};

window.applyCalcPreset = function(p){
    HIER_STATE.scaleMode = 'auto';
    HIER_STATE.scopeMode = 'cumulative';
    saveHierState();
    $('calc-code').value = 'rak_sub';
    $('ca-sub-demo').value = 'pdf';
    /* Presets are area-based: force the area-basis mode so the preset area always
       drives the engine (previously a preset clicked while in population mode set
       the hidden area but the engine kept recomputing from the population input). */
    if($('calc-mode') && $('calc-mode').value !== 'area'){ $('calc-mode').value = 'area'; calcModeChange(); }
    if(p === 'rak_cluster'){
        $('ca-mix-on').checked = false; toggleRefMix();
        $('calc-area-input').value = $('calc-unit').value === 'ha' ? 25 : 250000;
        $('calc-pct-res').value = 50; $('calc-pct-roads').value = 22; $('calc-pct-open').value = 12; $('calc-pct-fac').value = 10; $('calc-pct-com').value = 6;
        calcCodeChange(); $('rak-parcel-type').value = 'parcel_conv'; applyRakParcelPreset();
    } else if(p === 'rak_nh'){

function renderExpr(eq, isAr){
    var cls = isAr ? 'fm-eq-ar' : 'fm-eq-en';
    var dir = isAr ? 'rtl' : 'ltr';
    var inner = '';
    if(eq.type === 'frac'){
        inner = '<span class="fm-math">' +
            '<span>' + eq.lhs + '</span>' +
            '<span class="fm-op">=</span>' +
            '<span class="fm-frac"><span class="fm-num">' + eq.num + '</span><span class="fm-den">' + eq.den + '</span></span>' +
            (eq.mul ? '<span class="fm-op">' + eq.mul + '</span>' : '') +
        '</span>';
    } else {
        inner = '<span class="fm-math">' + eq.expr + '</span>';
    }
    return '<div class="' + cls + '" dir="' + dir + '">' + inner + '</div>';
}

function plainTextExpr(eq){
    if(eq.type === 'frac'){
        return eq.lhs + ' = (' + eq.num + ') / (' + eq.den + ')' + (eq.mul ? ' ' + eq.mul : '');

    return eq.expr;
}

var CONV = { m2: 1, ha: 10000, fed: 4200, ac: 4046.8564224, km2: 1000000, ft2: 0.09290304 };
window.fmConvertArea = function(srcKey){
    var val = parseFloat(document.getElementById('uc-' + srcKey).value);
    if(isNaN(val)) return;
    var m2 = val * CONV[srcKey];
    Object.keys(CONV).forEach(function(k){
        if(k === srcKey) return;
        var out = m2 / CONV[k];
        document.getElementById('uc-' + k).value = +out.toFixed(k === 'km2' ? 5 : (k === 'm2' || k === 'ft2' ? 1 : 4));
    });
};

var SR_KEY = 'um_sr_v18';
var srSession = null;
function umSrState(){ var s = lsGet(SR_KEY, {}); return (s && typeof s === 'object' && !Array.isArray(s)) ? s : {}; }
function umSrDeck(src){
    var deck = [];
    if(src !== 'enc' && typeof getAllFlashcards === 'function'){
        try{
            getAllFlashcards().forEach(function(c){
                deck.push({ key:'fc:' + c.id, cat:c.category || 'بطاقات', front:c.question, back:c.answer + (c.reference ? '\n\nالمرجع: ' + c.reference : '') });
            });
        }catch(e){}
    }
    if(src !== 'fc' && typeof DEFAULT_ENCYCLOPEDIA !== 'undefined' && Array.isArray(DEFAULT_ENCYCLOPEDIA)){
        DEFAULT_ENCYCLOPEDIA.forEach(function(t){
            deck.push({ key:'enc:' + t.id, cat:'مصطلحات: ' + (t.category || ''), front:'اشرح المصطلح: «' + t.arTerm + '»', back:t.definition + (t.application ? '\n\nالتطبيق: ' + t.application : '') + (t.enTerm ? '\n\nEN: ' + t.enTerm : '') });
        });
    }
    return deck;
}
function umSrDueCount(src){
    var st = umSrState(), now = Date.now(), n = 0;
    umSrDeck(src).forEach(function(c){ var r = st[c.key]; if(!r || !r.due || r.due <= now) n++; });
    return n;
}
function umSrRenderStats(){
    var el = $('um-sr-stats'), badge = $('um-sr-due-badge');
    if(!el) return;
    var src = $('um-sr-src') ? $('um-sr-src').value : 'all';
    var deck = umSrDeck(src), st = umSrState();
    var learned = 0;
    deck.forEach(function(c){ if(st[c.key] && st[c.key].reps > 0) learned++; });
    var due = umSrDueCount(src);
    el.textContent = 'البطاقات في النطاق: ' + deck.length + ' · متقنة أو قيد الإتقان: ' + learned + ' · مستحقة اليوم: ' + due;
    if(badge) badge.textContent = due ? ('مستحق اليوم: ' + due) : 'لا شيء مستحق اليوم';
}
window.umSrStart = function(){
    var src = $('um-sr-src') ? $('um-sr-src').value : 'all';
    var st = umSrState(), now = Date.now();
    var due = umSrDeck(src).filter(function(c){ var r = st[c.key]; return !r || !r.due || r.due <= now; });
    if(!due.length){ toast('لا توجد بطاقات مستحقة اليوم في هذا النطاق','amber'); return; }
    for(var i = due.length - 1; i > 0; i--){ var j = Math.floor(Math.random() * (i + 1)); var t = due[i]; due[i] = due[j]; due[j] = t; }
    srSession = { cards:due, idx:0, done:0, flipped:false };
    $('um-sr-session').classList.remove('hidden');
    umSrShowCard();
};
function umSrShowCard(){
    var c = srSession.cards[srSession.idx];
    $('um-sr-prog').textContent = 'بطاقة ' + (srSession.idx + 1) + ' من ' + srSession.cards.length;
    $('um-sr-cat').textContent = c.cat;
    $('um-sr-front').textContent = c.front;
    $('um-sr-back').classList.add('hidden');
    $('um-sr-back').textContent = c.back;
    $('um-sr-grades').classList.add('hidden');
    $('um-sr-flipbtn').classList.remove('hidden');
    srSession.flipped = false;
}
window.umSrFlip = function(){
    if(!srSession) return;
    $('um-sr-back').classList.remove('hidden');
    $('um-sr-grades').classList.remove('hidden');
    $('um-sr-flipbtn').classList.add('hidden');
    srSession.flipped = true;
};
window.umSrGrade = function(g){
    if(!srSession || !srSession.flipped) return;
    var c = srSession.cards[srSession.idx];
    var st = umSrState();
    var r = st[c.key] || { ivl:0, reps:0 };
    if(g === 0) r.ivl = 1;
    else if(g === 1) r.ivl = r.ivl >= 1 ? r.ivl * 2 : 3;
    else r.ivl = r.ivl >= 1 ? r.ivl * 3 : 7;
    r.reps = (r.reps || 0) + 1;
    r.due = Date.now() + r.ivl * DAY;
    st[c.key] = r;
    lsSet(SR_KEY, st);
    srSession.idx++; srSession.done++;
    if(srSession.idx >= srSession.cards.length){ umSrEnd(); return; }
    umSrShowCard();
};
window.umSrEnd = function(){
    if(srSession && srSession.done > 0) toast('أنهيت الجلسة بعد مراجعة ' + srSession.done + ' بطاقة');
    srSession = null;
    $('um-sr-session').classList.add('hidden');
    umSrRenderStats();
};

var MOCK_BEST_KEY = 'um_mock_best_v18';
var mock = null, mockTimer = null;
function umMockRenderBest(){
    var el = $('um-mock-best'); if(!el) return;
    var b = lsGet(MOCK_BEST_KEY, null);
    el.textContent = b ? ('أفضل نتيجة: ' + b.pct + '% (' + b.score + '/' + b.total + ') في ' + b.date) : '';
}
window.umMockStart = function(){
    if(typeof QUIZ_SCENARIOS === 'undefined' || !Array.isArray(QUIZ_SCENARIOS) || !QUIZ_SCENARIOS.length){ toast('بنك الأسئلة غير متاح','amber'); return; }
    var count = parseInt(($('um-mock-count') || {}).value, 10) || 10;
    var mins = parseInt(($('um-mock-mins') || {}).value, 10) || 10;
    var idxs = QUIZ_SCENARIOS.map(function(_, i){ return i; });
    for(var i = idxs.length - 1; i > 0; i--){ var j = Math.floor(Math.random() * (i + 1)); var t = idxs[i]; idxs[i] = idxs[j]; idxs[j] = t; }
    idxs = idxs.slice(0, Math.min(count, idxs.length));
    mock = { idxs:idxs, pos:0, answers:[], endsAt:Date.now() + mins * 60000 };
    $('um-mock-setup').classList.add('hidden');
    $('um-mock-results').classList.add('hidden');
    $('um-mock-results').innerHTML = '';
    $('um-mock-session').classList.remove('hidden');
    umMockShowQ();
    clearInterval(mockTimer);
    mockTimer = setInterval(umMockTick, 1000);
    umMockTick();
};
function umMockTick(){
    if(!mock){ clearInterval(mockTimer); return; }
    var rem = Math.max(0, mock.endsAt - Date.now());
    var mm = Math.floor(rem / 60000), ss = Math.floor((rem % 60000) / 1000);
    var el = $('um-mock-timer');
    if(el) el.textContent = 'الوقت المتبقي ' + ('0' + mm).slice(-2) + ':' + ('0' + ss).slice(-2);
    if(rem <= 0){ umMockFinish(false); toast('انتهى الوقت: تم تصحيح ما أجبت عنه','amber'); }
}
function umMockShowQ(){
    var q = QUIZ_SCENARIOS[mock.idxs[mock.pos]];
    $('um-mock-prog').textContent = 'سؤال ' + (mock.pos + 1) + ' من ' + mock.idxs.length;
    $('um-mock-cat').textContent = q.category || '';
    $('um-mock-q').textContent = q.question;
    $('um-mock-opts').innerHTML = q.options.map(function(opt, idx){
        return '<button onclick="umMockAnswer(' + idx + ')" class="w-full text-right p-3 rounded-xl bg-slate-800/50 hover:bg-slate-800 border border-slate-700 text-xs text-slate-200 transition flex items-start gap-2.5">' +
            '<span class="font-mono font-bold text-amber-400">' + (idx + 1) + '.</span><span>' + esc(opt) + '</span></button>';
    }).join('');
}
window.umMockAnswer = function(idx){
    if(!mock) return;
    mock.answers[mock.pos] = idx;
    mock.pos++;
    if(mock.pos >= mock.idxs.length){ umMockFinish(false); return; }
    umMockShowQ();
};
