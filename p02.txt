        function getAllFlashcards() {
            const full = [...appState.customFlashcards, ...DEFAULT_FLASHCARDS];
            const filtered = fcCatFilter === 'all' ? full : full.filter(c => c.category === fcCatFilter);
            if (fcShuffledOrder && fcShuffledOrder.length === filtered.length) {
                return fcShuffledOrder.map(idx => filtered[idx]).filter(Boolean);

        function populateFcCategories() {
            const sel = document.getElementById('fc-cat-filter');
            if (!sel) return;
            const full = [...appState.customFlashcards, ...DEFAULT_FLASHCARDS];
            const cats = ['all', ...Array.from(new Set(full.map(c => c.category)))];
            const cur = sel.value || fcCatFilter;
            sel.innerHTML = cats.map(c => `<option value="${c}" ${c === cur ? 'selected' : ''}>${c === 'all' ? 'كل التصنيفات (' + full.length + ')' : c}</option>`).join('');
        }

        function filterFlashcardsByCat() {
            const sel = document.getElementById('fc-cat-filter');
            fcCatFilter = sel ? sel.value : 'all';
            fcShuffledOrder = null;
            appState.currentFlashcardIndex = 0;
            renderFlashcard();
        }

        function shuffleFlashcards() {
            const full = [...appState.customFlashcards, ...DEFAULT_FLASHCARDS];
            const filtered = fcCatFilter === 'all' ? full : full.filter(c => c.category === fcCatFilter);
            const arr = filtered.map((_, i) => i);
            for (let i = arr.length - 1; i > 0; i--) {
                const j = Math.floor(Math.random() * (i + 1));
                [arr[i], arr[j]] = [arr[j], arr[i]];
            }
            fcShuffledOrder = arr;
            appState.currentFlashcardIndex = 0;
            renderFlashcard();
            showToast('تم خلط البطاقات عشوائياً!', 'amber');
        }

        function renderFlashcard() {
            populateFcCategories();
            const deck = getAllFlashcards();
            if (deck.length === 0) return;
            if (appState.currentFlashcardIndex >= deck.length) appState.currentFlashcardIndex = 0;
            if (appState.currentFlashcardIndex < 0) appState.currentFlashcardIndex = deck.length - 1;

            const card = deck[appState.currentFlashcardIndex];
            const isMastered = !!appState.flashcardMastery[card.id];
            const totalAll = [...appState.customFlashcards, ...DEFAULT_FLASHCARDS];
            const masteredCount = totalAll.filter(c => appState.flashcardMastery[c.id]).length;
            const mStat = document.getElementById('fc-mastered-stat');
            if (mStat) mStat.textContent = `أتقنت: ${masteredCount}/${totalAll.length}`;

            // Reset flip state
            const inner = document.getElementById('fc-card-inner');
            inner.classList.remove('rotate-y-180');

            document.getElementById('fc-counter').textContent = `بطاقة ${appState.currentFlashcardIndex + 1} من ${deck.length}`;
            const badge = document.getElementById('fc-mastery-badge');
            if (isMastered) {
                badge.className = 'px-2.5 py-0.5 rounded-full bg-emerald-500/20 text-emerald-300 border border-emerald-500/30 font-bold';
                badge.textContent = 'متقنة ✓';
            } else {
                badge.className = 'px-2.5 py-0.5 rounded-full bg-amber-500/20 text-amber-300 border border-amber-500/30 font-bold';
                badge.textContent = 'قيد المراجعة';
            }

            document.getElementById('fc-front-cat').textContent = card.category;
            document.getElementById('fc-front-question').textContent = card.question;
            document.getElementById('fc-back-answer').innerHTML = card.answer.replace(/\n/g, '<br>');
            document.getElementById('fc-back-ref').textContent = 'المرجع: ' + card.reference;
        }

        function checkQuizAnswer(selectedIdx) {
            const qIdx = appState.currentQuizIndex;
            const q = QUIZ_SCENARIOS[qIdx];
            const fb = document.getElementById('quiz-feedback');
            fb.classList.remove('hidden');

            // Highlight options
            q.options.forEach((_, idx) => {
                const btn = document.getElementById(`qopt-${idx}`);
                if (!btn) return;
                if (idx === q.correctIndex) {

function buildUnifiedRefEntries(){
    var seenEn = {};
    var out = [];
    // 1. User custom concepts first
    (appState.customConcepts || []).forEach(function(item){
        out.push({ ar: item.arTerm, en: item.enTerm, cat: item.category, src: item.source, def: item.definition, app: item.application, deep: true });
        seenEn[(item.enTerm || '').toLowerCase()] = true;
    });
    // 2. Deep encyclopedia entries
    DEFAULT_ENCYCLOPEDIA.forEach(function(item){
        out.push({ ar: item.arTerm, en: item.enTerm, cat: item.category, src: item.source, def: item.definition, app: item.application, deep: true });
        seenEn[(item.enTerm || '').toLowerCase()] = true;
    });
    // 3. Unique extra glossary entries (zero overlap)
    EXTRA_GLOSSARY.forEach(function(g){
        if(seenEn[g[1].toLowerCase()]) return;
        out.push({ ar: g[0], en: g[1], cat: g[2], src: 'القاموس التخطيطي الموحّد', def: g[3], app: '', deep: false });
    });
    return out;
}

window.renderUnifiedReference = function(){
    var all = buildUnifiedRefEntries();
    var cats = ['الكل'].concat(all.map(function(x){ return x.cat; }).filter(function(v,i,a){ return a.indexOf(v) === i; }));
    var pills = document.getElementById('enc-category-pills');
    if(pills){
        pills.innerHTML = cats.map(function(c){
            var on = (window._refCat === c);
            return '<button onclick="setEncCategory(this.dataset.c)" data-c="'+c+'" class="um-chip '+(on?'on':'')+'">'+c+'</button>';
        }).join('');
    }
    var q = (document.getElementById('enc-search-input') ? document.getElementById('enc-search-input').value : '').trim().toLowerCase();
    var filtered = all.filter(function(x){
        if(window._refCat !== 'الكل' && x.cat !== window._refCat) return false;
        if(!q) return true;
        return (x.ar + ' ' + x.en + ' ' + x.def + ' ' + x.app + ' ' + x.src).toLowerCase().indexOf(q) !== -1;
    });
    var cntEl = document.getElementById('ref-dict-count');
    if(cntEl) cntEl.textContent = filtered.length + ' مصطلح ومفهوم';
    var grid = document.getElementById('encyclopedia-grid');
    if(!grid) return;
    if(!filtered.length){
        grid.innerHTML = '<div class="col-span-2 p-6 text-center rounded-xl bg-slate-900/70 border border-slate-800 text-xs text-slate-400">لا توجد نتائج مدورة لبحثك. يمكنك إضافة مصطلح جديد عبر الزر بالأعلى.</div>';
        return;
    }
    grid.innerHTML = filtered.map(function(item){
        return '<details class="ref-card-det">' +
            '<summary>' +
                '<div class="space-y-0.5">' +
                    '<div class="font-extrabold text-sm sm:text-base" style="color:var(--h)">' + item.ar + '</div>' +
                    '<div class="um-en" style="text-align:right">' + item.en + '</div>' +
                '</div>' +
                '<span class="um-tag">' + item.cat + '</span>' +
            '</summary>' +
            '<div class="ref-card-body space-y-2.5">' +
                '<p style="margin:0">' + item.def + '</p>' +
                (item.app ? '<div class="p-3 rounded-xl bg-slate-950/60 border border-slate-800 text-xs space-y-1"><div class="font-bold text-teal-400"><i class="fa-solid fa-compass-drafting ml-1"></i>التطبيق العملي في المخططات:</div><p style="margin:0">' + item.app + '</p></div>' : '') +
                '<div class="text-[11px] font-mono opacity-75"><i class="fa-solid fa-bookmark text-amber-400 ml-1"></i>' + item.src + '</div>' +
            '</div>' +
        '</details>';
    }).join('');
    var _fp = document.getElementById('tab-formulas');
    if (_fp && _fp.umPinHeaders) { _fp.umPinHeaders(); }
};

function renderChecklist(){
    var ctn = document.getElementById('checklist-container');
    if(!ctn) return;
    var st = {};
    try { st = JSON.parse(localStorage.getItem('um_ck') || '{}'); } catch(e){}
    var total = 0, done = 0;
    var html = CK.map(function(g, i){
        var gDone = 0;
        var itemsHtml = g[1].map(function(x, j){
            var k = i + '-' + j;
            total++;
            if(st[k]){ done++; gDone++; }
            return '<label><input type="checkbox" '+(st[k]?'checked':'')+' onchange="ckSave(\''+k+'\',this.checked)"><span>'+x+'</span></label>';
        }).join('');
        return '<details class="um-det" '+(i===0?'open':'')+'><summary><span>'+g[0]+'</span><span class="um-tag" style="margin-inline-start:.5rem">'+gDone+'/'+g[1].length+'</span></summary><div class="um-in um-ck">'+itemsHtml+'</div></details>';
    }).join('');
    ctn.innerHTML = html;
    var pill = document.getElementById('ck-progress-pill');
    if(pill) pill.textContent = Math.round(total ? (done/total*100) : 0) + '% (' + done + '/' + total + ')';
}
window.ckSave = function(k, v){
    var s = {};
    try { s = JSON.parse(localStorage.getItem('um_ck') || '{}'); } catch(e){}
    s[k] = v;
    try { localStorage.setItem('um_ck', JSON.stringify(s)); } catch(e){}
    renderChecklist();
};

function greedyAlloc(demand, minThresh, tiers){
    var counts = {};
    tiers.forEach(function(t){ counts[t.key] = 0; });
    var rem = demand, guard = 0;
    while(rem >= minThresh && guard < 100){
        guard++;
        var chosen = tiers[tiers.length - 1];
        for(var i = 0; i < tiers.length; i++){
            if(rem <= tiers[i].cap){ chosen = tiers[i]; break; }
        }
        counts[chosen.key]++;
        rem -= chosen.cap;
    }
    return counts;
}
function greedyClinics(pop, tiers, minRem){
    var counts = {};
    tiers.forEach(function(t){ counts[t.key] = 0; });
    var rem = pop, guard = 0;
    while(rem >= tiers[0].min && guard < 50){
        guard++;
        var chosen = tiers[tiers.length - 1];
        for(var i = 0; i < tiers.length; i++){
            var upper = (i < tiers.length - 1) ? tiers[i+1].min : Infinity;
            if(rem >= tiers[i].min && rem < upper){ chosen = tiers[i]; break; }
        }
        counts[chosen.key]++;
        rem -= chosen.max;
        if(rem < minRem) break;
    }
    return counts;
}

function detectPlanningScale(codeKey, ha, pop, isPopOnly){
    var r = SCALE_RULES[codeKey] || SCALE_RULES.rak_sub;
    var popTier = pop < r.t1.maxPop ? 1 : (pop <= r.t2.maxPop ? 2 : (pop <= r.t3.maxPop ? 3 : 4));
    var areaTier = ha < r.t1.maxHa ? 1 : (ha <= r.t2.maxHa ? 2 : (ha <= r.t3.maxHa ? 3 : 4));
    var finalTier = isPopOnly ? popTier : Math.max(popTier, areaTier);
    var mismatch = (!isPopOnly && popTier !== areaTier)
        ? ('ملاحظة كثافة: تصنيف المساحة (' + ha.toFixed(1) + ' هكتار) يعطي مستوى «' + TIER_META[areaTier].ar + '» بينما تصنيف السكان (' + Math.round(pop).toLocaleString('en') + ' نسمة) يعطي مستوى «' + TIER_META[popTier].ar + '» — تم اعتماد المستوى الأعلى لضمان تغطية نطاقات المسافة والسكان معاً.')
        : '';
    return { tier: finalTier, popTier: popTier, areaTier: areaTier, desc: r['t' + finalTier].desc, mismatch: mismatch };
}

window.runSettlementScorer = function(){
    var d1 = +$('sc-den-ex').value || 1;
    var d2 = +$('sc-den-pr').value || 1;
    var fc = +$('sc-fac').value || 1;
    var ge = +$('sc-geo').value || 1;
    var score = (d1*0.125) + (d2*0.125) + (fc*0.35) + (ge*0.40);
    var label, codeKey, badgeCls;
    if(score <= 2.4){
        label = 'ريفي (Rural Settlement)'; codeKey = 'rak_rur';
        badgeCls = 'bg-emerald-500/20 text-emerald-300 border-emerald-500/30';
    } else if(score <= 3.6){
        label = 'ضواحي (Suburban Settlement)'; codeKey = 'rak_sub';
        badgeCls = 'bg-teal-500/20 text-teal-300 border-teal-500/30';
    } else {
        label = 'حضري (Urban Settlement)'; codeKey = 'rak_urb';
        badgeCls = 'bg-amber-500/20 text-amber-300 border-amber-500/30';
    }
    window._scorerCode = codeKey;
    var b = $('scorer-badge');
    if(b){
        b.className = 'text-xs font-mono px-2.5 py-0.5 rounded-full border shrink-0 ' + badgeCls;
        b.textContent = label + ' (' + score.toFixed(2) + ')';
    }
    var txt = $('scorer-result-text');
    if(txt){
        txt.innerHTML = 'النتيجة الموزونة: <span class="font-mono text-amber-300">' + score.toFixed(2) + ' / 5.00</span> ← التصنيف المعتمد: <span class="text-teal-300">' + label + '</span>';
    }
};

};
